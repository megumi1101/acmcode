vector<int> parent(n + 5, -1);
vector<int> loop;
bool fg = 0;
function<void(int, int)> dfs = [&](int u, int p) {
    parent[u] = p;
    for (int v : ed[u]) {
        if (fg) return;
        if (v == p) continue;  
        if (parent[v] != -1) { 
            int x = u;
            fg = 1;
            while (x) {
                loop.push_back(x);
                if (x == v) break;
                x = parent[x];
            }
            return;
        }
        dfs(v, u);
    }
};
