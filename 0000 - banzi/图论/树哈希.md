需要找到树的中心,多个中心，两个树任意一个的中心相等就可以

如果是基环树同构,找到环，把环的点的hash值求出来，
然后把一个的hash序列长度*2, 正反做两遍hash就可以,因为环可以空间翻转同构
```cpp
constexpr int Bas = 257;
auto getH = [&](ull x) {
    return x * x * x * 998244389 + 998244391;
};
auto getF = [&](ull x) {
    return getH(x & ((1ll << 32) - 1)) + getH(x >> 32);
};
auto dfs = [&](auto &&dfs, int u, int fat) -> void {
    h[u] = Bas;
    for (auto v : ed[u]) {
        if (v == fat) continue;
        dfs(dfs, v, u);
        h[u] += getF(h[v]);
    }
};
dfs(dfs, loop[i], 0);
```
