#include <bits/stdc++.h>
using namespace std;

#define ls (u << 1)
#define rs (u << 1 | 1)

/*
 * 1. 单点修改 + 区间查询
 *
 * n 表示最大下标
 * 下标范围：[0, n]
 * 若由 vector<Info> a 建树，则 n = a.size() - 1
 *
 * query(l, r) / findFirst(l, r) / findLast(l, r)
 * 均表示闭区间 [l, r]
 *
 * 要求 Info 支持：
 *   Info{}
 *   Info operator+(const Info&, const Info&)
 */
template<typename Info>
struct SegmentTree {
    int n;
    vector<Info> info;

    SegmentTree(int n = -1) : n(n), info(n >= 0 ? 4 * (n + 1) + 5 : 0) {}

    SegmentTree(const vector<Info> &a) : SegmentTree((int)a.size() - 1) {
        if (n >= 0) build(1, 0, n, a);
    }

    void pull(int u) {
        info[u] = info[ls] + info[rs];
    }

    void build(int u, int l, int r, const vector<Info> &a) {
        if (l == r) return void(info[u] = a[l]);
        int m = (l + r) >> 1;
        build(ls, l, m, a);
        build(rs, m + 1, r, a);
        pull(u);
    }

    void build(const vector<Info> &a) {
        n = (int)a.size() - 1;
        info.assign(n >= 0 ? 4 * (n + 1) + 5 : 0, Info{});
        if (n >= 0) build(1, 0, n, a);
    }

    void modify(int u, int l, int r, int x, const Info &v) {
        if (l == r) return void(info[u] = v);
        int m = (l + r) >> 1;
        x <= m ? modify(ls, l, m, x, v) : modify(rs, m + 1, r, x, v);
        pull(u);
    }

    void modify(int x, const Info &v) {
        assert(0 <= x && x <= n);
        modify(1, 0, n, x, v);
    }

    Info query(int L, int R, int u, int l, int r) {
        if (L <= l && r <= R) return info[u];
        int m = (l + r) >> 1;
        if (R <= m) return query(L, R, ls, l, m);
        if (L > m) return query(L, R, rs, m + 1, r);
        return query(L, R, ls, l, m) + query(L, R, rs, m + 1, r);
    }

    Info query(int l, int r) {
        assert(0 <= l && l <= r && r <= n);
        return query(l, r, 1, 0, n);
    }

    template<class F>
    int findFirst(int L, int R, F check, int u, int l, int r) {
        if (r < L || R < l || !check(info[u])) return -1;
        if (l == r) return l;
        int m = (l + r) >> 1;
        int p = findFirst(L, R, check, ls, l, m);
        return p != -1 ? p : findFirst(L, R, check, rs, m + 1, r);
    }

    template<class F>
    int findFirst(int l, int r, F check) {
        if (l > r) return -1;
        return findFirst(l, r, check, 1, 0, n);
    }

    template<class F>
    int findLast(int L, int R, F check, int u, int l, int r) {
        if (r < L || R < l || !check(info[u])) return -1;
        if (l == r) return l;
        int m = (l + r) >> 1;
        int p = findLast(L, R, check, rs, m + 1, r);
        return p != -1 ? p : findLast(L, R, check, ls, l, m);
    }

    template<class F>
    int findLast(int l, int r, F check) {
        if (l > r) return -1;
        return findLast(l, r, check, 1, 0, n);
    }
};

/*
 * 2. 区间修改 + 区间查询（懒标记）
 *
 * n 表示最大下标
 * 下标范围：[0, n]
 * 若由 vector<Info> a 建树，则 n = a.size() - 1
 *
 * modify(l, r) / query(l, r) / findFirst / findLast
 * 均表示闭区间 [l, r]
 *
 * 要求：
 *   Info{}
 *   Tag{}
 *   Info operator+(const Info&, const Info&)
 *   info.apply(tag)
 *   tag.apply(tag)
 *
 * Tag{} 必须表示空标记 / 单位标记。
 */
template<typename Info, typename Tag>
struct LazySegmentTree {
    int n;
    vector<Info> info;
    vector<Tag> tag;

    LazySegmentTree(int n = -1)
        : n(n),
          info(n >= 0 ? 4 * (n + 1) + 5 : 0),
          tag(n >= 0 ? 4 * (n + 1) + 5 : 0) {}

    LazySegmentTree(const vector<Info> &a)
        : LazySegmentTree((int)a.size() - 1) {
        if (n >= 0) build(1, 0, n, a);
    }

    void pull(int u) {
        info[u] = info[ls] + info[rs];
    }

    void apply(int u, const Tag &v) {
        info[u].apply(v);
        tag[u].apply(v);
    }

    void push(int u) {
        apply(ls, tag[u]);
        apply(rs, tag[u]);
        tag[u] = Tag{};
    }

    void build(int u, int l, int r, const vector<Info> &a) {
        if (l == r) return void(info[u] = a[l]);
        int m = (l + r) >> 1;
        build(ls, l, m, a);
        build(rs, m + 1, r, a);
        pull(u);
    }

    void build(const vector<Info> &a) {
        n = (int)a.size() - 1;
        info.assign(n >= 0 ? 4 * (n + 1) + 5 : 0, Info{});
        tag.assign(n >= 0 ? 4 * (n + 1) + 5 : 0, Tag{});
        if (n >= 0) build(1, 0, n, a);
    }

    void modify(int L, int R, int u, int l, int r, const Tag &v) {
        if (L <= l && r <= R) return apply(u, v);
        push(u);
        int m = (l + r) >> 1;
        if (L <= m) modify(L, R, ls, l, m, v);
        if (R > m) modify(L, R, rs, m + 1, r, v);
        pull(u);
    }

    void modify(int l, int r, const Tag &v) {
        assert(0 <= l && l <= r && r <= n);
        modify(l, r, 1, 0, n, v);
    }

    Info query(int L, int R, int u, int l, int r) {
        if (L <= l && r <= R) return info[u];
        push(u);
        int m = (l + r) >> 1;
        if (R <= m) return query(L, R, ls, l, m);
        if (L > m) return query(L, R, rs, m + 1, r);
        return query(L, R, ls, l, m) + query(L, R, rs, m + 1, r);
    }

    Info query(int l, int r) {
        assert(0 <= l && l <= r && r <= n);
        return query(l, r, 1, 0, n);
    }

    template<class F>
    int findFirst(int L, int R, F check, int u, int l, int r) {
        if (r < L || R < l || !check(info[u])) return -1;
        if (l == r) return l;
        push(u);
        int m = (l + r) >> 1;
        int p = findFirst(L, R, check, ls, l, m);
        return p != -1 ? p : findFirst(L, R, check, rs, m + 1, r);
    }

    template<class F>
    int findFirst(int l, int r, F check) {
        if (l > r) return -1;
        return findFirst(l, r, check, 1, 0, n);
    }

    template<class F>
    int findLast(int L, int R, F check, int u, int l, int r) {
        if (r < L || R < l || !check(info[u])) return -1;
        if (l == r) return l;
        push(u);
        int m = (l + r) >> 1;
        int p = findLast(L, R, check, rs, m + 1, r);
        return p != -1 ? p : findLast(L, R, check, ls, l, m);
    }

    template<class F>
    int findLast(int l, int r, F check) {
        if (l > r) return -1;
        return findLast(l, r, check, 1, 0, n);
    }
};

#undef ls
#undef rs
