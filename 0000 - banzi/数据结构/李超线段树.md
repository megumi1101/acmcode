// ======================= Li Chao Segment Tree (Min) =======================
// 作用：维护一组直线 y = kx + b，支持：add_line / add_segment / query(x)
// 复杂度：插入/查询 O(logX)（X 为横坐标范围）
// 注意：本实现是“最小值版”。若要求最大值，可插入时对 (k,b) 取反，查询后再取反。
struct LiChao {
    struct Line {
        long long k, b; // y = kx + b
        Line(long long _k=0, long long _b=(long long)4e18): k(_k), b(_b) {}
        inline long long get(long long x) const {
            return (long long)((__int128)k * x + b); // 防中间溢出
        }
    };
    struct Node {
        Line ln;
        Node *lch, *rch;
        Node(): ln(), lch(nullptr), rch(nullptr) {}
    };

    long long X_LO, X_HI; // 维护的 x 范围（闭区间）
    Node* root;

    // 构造：给定 x 的范围（默认 1..1e9）
    LiChao(long long lo=1, long long hi=1000000000LL): X_LO(lo), X_HI(hi) {
        root = nullptr;
    }

    // 清空
    void clear() { destroy(root); root = nullptr; }
    ~LiChao(){ destroy(root); }

    // 插入整条直线 y = kx + b （在 [X_LO, X_HI] 整个区间有效）
    void add_line(long long k, long long b) { add_line(Line(k,b), root, X_LO, X_HI); }

    // 插入线段（L..R）上有效的直线 y = kx + b
    void add_segment(long long L, long long R, long long k, long long b) {
        if (L > R) return;
        add_segment(Line(k,b), root, X_LO, X_HI, L, R);
    }

    // 单点查询：给定 x，返回 min y
    long long query(long long x) const { return query(x, root, X_LO, X_HI); }

private:
    static inline long long mid(long long l, long long r){ return l + ((r - l) >> 1); }

    void destroy(Node* p){
        if (!p) return;
        destroy(p->lch);
        destroy(p->rch);
        delete p;
    }

    // 在区间 [l,r] 上插入整条直线 nl
    void add_line(Line nl, Node* &p, long long l, long long r){
        if (!p) { p = new Node(); p->ln = nl; return; }
        long long m = mid(l, r);

        // 当前点处记录“更优”的直线于 p->ln
        bool left = nl.get(l) < p->ln.get(l);
        bool midv = nl.get(m) < p->ln.get(m);
        if (midv) swap(nl, p->ln);

        if (l == r) return;
        if (left != midv) add_line(nl, p->lch, l, m);
        else add_line(nl, p->rch, m+1, r);
    }

    // 在区间 [ql,qr] 上插入线段直线 nl
    void add_segment(Line nl, Node* &p, long long l, long long r, long long ql, long long qr){
        if (qr < l || r < ql) return;
        if (ql <= l && r <= qr) { add_line(nl, p, l, r); return; }
        long long m = mid(l, r);
        if (!p) p = new Node();
        add_segment(nl, p->lch, l, m, ql, qr);
        add_segment(nl, p->rch, m+1, r, ql, qr);
    }

    long long query(long long x, Node* p, long long l, long long r) const {
        if (!p) return (long long)4e18;
        long long res = p->ln.get(x);
        if (l == r) return res;
        long long m = mid(l, r);
        if (x <= m) return min(res, query(x, p->lch, l, m));
        else        return min(res, query(x, p->rch, m+1, r));
    }
};
// ===================== end of Li Chao Segment Tree =====================
